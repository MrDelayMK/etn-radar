// ---------- Rahmen-Werkzeug ----------
// Derselbe Rahmen wie im Bau-Skript (scripts/share-bilder.mjs): Linie in der
// Farbe mit weichen Luecken oben links fuer Logo + Name und unten rechts fuer
// die Domain, Schrift mit dunklem Halo. Ergebnis als <img>, weil diese Seite
// selbst keine Downloads ausloesen darf.
const WZ = { bild: null, farbe: "#5b9cff" };
const WZ_FARBEN = GRUPPEN.filter((g) => g.hex)
  .map((g) => [g.hex, g.titel])
  .filter((f, i, a) => a.findIndex((x) => x[0] === f[0]) === i);

function wzFarbenZeigen() {
  document.getElementById("wzFarben").innerHTML =
    WZ_FARBEN.map(([hex, name]) =>
      '<button type="button" style="--c:' + hex + '" title="' + esc(name) + '" aria-label="' + esc(name) +
      '" aria-pressed="' + (hex === WZ.farbe) + '" data-farbe="' + hex + '"></button>').join("") +
    '<input type="color" id="wzEigene" value="' + WZ.farbe + '" title="Eigene Farbe" aria-label="Eigene Farbe">';
}

const wzHell = (hex, anteil) => {
  const n = parseInt(hex.slice(1), 16);
  const k = [(n >> 16) & 255, (n >> 8) & 255, n & 255].map((c) => Math.round(c + (255 - c) * anteil));
  return "rgb(" + k.join(",") + ")";
};

function wzBauen() {
  if (!WZ.bild || !LOGO_BILD.complete) return;
  const [W, H] = document.getElementById("wzFormat").value.split("x").map(Number);
  // Wie im Skript: das breite Format bekommt die kleinere Variante.
  const s = H < 1000 ? 0.8 : 1;
  const i = Math.round(46 * s), r = Math.round(30 * s), logoS = Math.round(64 * s);
  const luft = 16 * s, verlauf = 28 * s, zurEcke = 56 * s, abstand = Math.round(logoS * 0.25);

  const c = document.createElement("canvas");
  c.width = W; c.height = H;
  const x = c.getContext("2d");

  // Bild wie object-fit: cover
  const b = WZ.bild;
  const f = Math.max(W / b.naturalWidth, H / b.naturalHeight);
  x.drawImage(b, (W - b.naturalWidth * f) / 2, (H - b.naturalHeight * f) / 2, b.naturalWidth * f, b.naturalHeight * f);

  const nameFont = "700 " + Math.round(34 * s) + "px 'Segoe UI', system-ui, sans-serif";
  const domFont = "700 " + Math.round(22 * s) + "px Consolas, ui-monospace, monospace";
  const domText = "etn-radar.galacticsl.com";
  x.font = nameFont; const nameW = x.measureText("ETN Radar").width;
  x.font = domFont; const domW = x.measureText(domText).width;

  const obenVon = i + r + zurEcke, obenBis = obenVon + logoS + abstand + nameW;
  const untenBis = W - i - r - zurEcke, untenVon = untenBis - domW;

  // Linie auf eigener Ebene, dann die Luecken mit weichem Verlauf herausnehmen
  const l = document.createElement("canvas");
  l.width = W; l.height = H;
  const lx = l.getContext("2d");
  const rahmen = (breite) => { lx.beginPath(); lx.roundRect(i, i, W - 2 * i, H - 2 * i, r); lx.lineWidth = breite; lx.stroke(); };
  lx.save(); lx.filter = "blur(7px)"; lx.globalAlpha = 0.7; lx.strokeStyle = WZ.farbe; rahmen(8); lx.restore();
  lx.strokeStyle = wzHell(WZ.farbe, 0.35); rahmen(3);
  lx.globalCompositeOperation = "destination-out";
  const luecke = (von, bis, y) => {
    const x0 = von - luft - verlauf, x1 = bis + luft + verlauf;
    const g = lx.createLinearGradient(x0, 0, x1, 0);
    const t = verlauf / (x1 - x0);
    g.addColorStop(0, "rgba(0,0,0,0)"); g.addColorStop(t, "rgba(0,0,0,1)");
    g.addColorStop(1 - t, "rgba(0,0,0,1)"); g.addColorStop(1, "rgba(0,0,0,0)");
    lx.fillStyle = g; lx.fillRect(x0, y - 30, x1 - x0, 60);
  };
  luecke(obenVon, obenBis, i);
  luecke(untenVon, untenBis, H - i);

  // Dunkle weiche Flaechen hinter Marke und Domain, dann die Linie darueber
  x.save(); x.filter = "blur(16px)"; x.fillStyle = "rgba(6,16,31,.6)";
  x.beginPath(); x.roundRect(obenVon - 12, i - logoS / 2 - 4, obenBis - obenVon + 24, logoS + 8, logoS / 2); x.fill();
  x.beginPath(); x.roundRect(untenVon - 14, H - i - 20 * s, domW + 28, 40 * s, 20 * s); x.fill();
  x.restore();
  x.drawImage(l, 0, 0);

  // Schrift mit Halo, zweimal gezeichnet wie im Skript
  const text = (t, font, tx, ty) => {
    x.save(); x.font = font; x.textBaseline = "middle";
    x.shadowColor = "rgba(2,6,17,.95)"; x.shadowBlur = 10; x.fillStyle = "#fff";
    x.fillText(t, tx, ty); x.fillText(t, tx, ty); x.restore();
  };
  x.drawImage(LOGO_BILD, obenVon, i - logoS / 2, logoS, logoS);
  text("ETN Radar", nameFont, obenVon + logoS + abstand, i + 1);
  text(domText, domFont, untenVon, H - i + 1);

  const img = new Image();
  img.alt = "Gerahmtes Bild";
  img.src = c.toDataURL("image/jpeg", 0.9);
  const box = document.getElementById("wzErgebnis");
  box.innerHTML = "";
  box.appendChild(img);
  document.getElementById("wzHinweis").hidden = false;
}

const LOGO_BILD = new Image();
LOGO_BILD.onload = () => wzBauen();
LOGO_BILD.src = document.querySelector(".marke img")?.getAttribute?.("src") ?? "";

function wzLaden(datei) {
  if (!datei || !/^image\//.test(datei.type)) return;
  const b = new Image();
  b.onload = () => {
    WZ.bild = b;
    document.getElementById("wzAblageText").textContent = datei.name + " - anderes Bild? Einfach nochmal";
    wzBauen();
  };
  b.src = URL.createObjectURL(datei);
}

wzFarbenZeigen();
document.getElementById("wzFarben").addEventListener("click", (e) => {
  const k = e.target.closest("[data-farbe]");
  if (!k) return;
  WZ.farbe = k.dataset.farbe;
  wzFarbenZeigen();
  wzBauen();
});
document.getElementById("wzFarben").addEventListener("input", (e) => {
  if (e.target.id !== "wzEigene") return;
  WZ.farbe = e.target.value;
  document.querySelectorAll("#wzFarben [data-farbe]").forEach((k) => k.setAttribute("aria-pressed", "false"));
  wzBauen();
});
document.getElementById("wzFormat").addEventListener("change", wzBauen);
document.getElementById("wzDatei").addEventListener("change", (e) => wzLaden(e.target.files[0]));

ablage.addEventListener("dragover", (e) => { e.preventDefault(); ablage.classList.add("drueber"); });
ablage.addEventListener("dragleave", () => ablage.classList.remove("drueber"));
ablage.addEventListener("drop", (e) => { e.preventDefault(); ablage.classList.remove("drueber"); wzLaden(e.dataTransfer.files[0]); });
